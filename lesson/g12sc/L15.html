    \(\displaystyle \implies I = \int udv = uv - \int vdu = x\arcsin x - \int \frac{x}{\sqrt{1-x^{2}}}dx = x\arcsin x - J\), \(\displaystyle \left(J = \int \frac{x}{\sqrt{1-x^{2}}}dx\right)\)
    គណនា \(\displaystyle J = \int \frac{x}{\sqrt{1-x^{2}}}dx\) (ប្រើរូបមន្តប្តូរអថេរ)

    តាង \(\displaystyle \begin{cases} u = \arctan x \implies du = \frac{dx}{1+x^{2}} \\ dv = dx \implies v = x \end{cases}\)

    គណនា \(\displaystyle J = \int \frac{x^{2}}{\sqrt{1-x^{2}}}dx\)
    តាង \(\displaystyle x = \sin t \implies dx = \cos tdt\) \(\displaystyle (t = \arcsin x)\)
    គេបាន

    \[\begin{aligned} J &= \int \frac{\sin^{2}t\cos tdt}{\sqrt{1-\sin^{2}t}} = \int \sin^{2}tdt \\ &= \frac{1}{2}\int (1 - \cos 2t)dt = \frac{1}{2}t - \frac{1}{4}\sin 2t + c_{1} \\ &= \frac{1}{2}\arcsin x - \frac{1}{2}\sin t\cos t + c_{1} \\ &= \frac{1}{2}\arcsin x - \frac{1}{2}x\sqrt{1-x^{2}} + c_{1} \end{aligned}\]

    យក \(\displaystyle J\) ជំនួសក្នុង \(\displaystyle I\) គេបាន៖ \(\displaystyle I = \frac{1}{2}x^{2}\arccos x + \frac{1}{4}\arcsin x - \frac{1}{4}x\sqrt{1-x^{2}} + c, c\) ជាចំនួនពិតថេរ ។

  4. ឃ.

    \(\displaystyle I = \int x\operatorname{arccot} xdx\)
    តាង \(\displaystyle \begin{cases} u = \operatorname{arccot} x \implies du = -\frac{1}{1+x^{2}}dx \\ dv = xdx \implies v = \frac{1}{2}x^{2} \end{cases}\)

    \[\begin{aligned} \implies I = \int udv = uv - \int vdu &= \frac{1}{2}x^{2}\operatorname{arccot} x + \frac{1}{2}\int \frac{x^{2}}{1+x^{2}}dx \\ &= \frac{1}{2}x^{2}\operatorname{arccot} x + \frac{1}{2}\int \frac{x^{2}+1-1}{x^{2}+1}dx \\ &= \frac{1}{2}x^{2}\operatorname{arccot} x + \frac{1}{2}\int dx - \frac{1}{2}\int \frac{dx}{x^{2}+1} \\ &= \frac{1}{2}x^{2}\operatorname{arccot} x + \frac{1}{2}x - \frac{1}{2}\arctan x + c \end{aligned}\]

    យកទៅជួសចូលក្នុង \(\displaystyle I\) គេបាន ៖ \(\displaystyle I = \frac{x^{2}}{2}\operatorname{arccot} x + \frac{1}{2}x - \frac{1}{2}\arctan x + c, c\) ជាចំនួនពិតថេរ ។

ឧទាហរណ៍ 1 : គណនា \(\displaystyle \frac{2}{x+1} + \frac{3}{x-3} = \frac{2(x-3)+3(x+1)}{(x+1)(x-3)} = \frac{2x-6+3x+3}{(x+1)(x-3)} = \frac{5x-3}{(x+1)(x-3)}\) ដោយបូកប្រភាគនៅអង្គខាងឆ្វេង គេបានប្រភាគនៅអង្គខាងស្តាំ ។ ច្រាសមកវិញរកចំនួនពិតថេរ \(\displaystyle A\) និង \(\displaystyle B\) ដោយដឹងថា \(\displaystyle \frac{5x-3}{(x+1)(x-3)} = \frac{A}{x+1} + \frac{B}{x-3}\) \(\displaystyle \frac{5x-3}{(x+1)(x-3)} = \frac{A}{x+1} + \frac{B}{x-3} = \frac{A(x-3)+B(x+1)}{(x+1)(x-3)} = \frac{(A+B)x - 3A + B}{(x+1)(x-3)}\) គេបាន \(\displaystyle 5x - 3 = (A+B)x - 3A + B\) ដោយផ្ទឹមលេខមេគុណ គេបាន \(\displaystyle \begin{cases} A+B=5 \\ -3A+B=-3 \end{cases}\) គេបាន \(\displaystyle A=2, B=3\) នាំឱ្យ \(\displaystyle \frac{5x-3}{(x+1)(x-3)} = \frac{2}{x+1} + \frac{3}{x-3}\) ។

ឧទាហរណ៍ 2 : គណនាអាំងតេក្រាល \(\displaystyle \int \frac{5x-3}{(x+1)(x-3)}dx\) លំនាំដូចគ្នាឧទាហរណ៍ 1 គេបាន

\[\begin{aligned} \int \frac{5x-3}{(x+1)(x-3)}dx &= \int \left[\frac{2}{x+1} + \frac{3}{x-3}\right]dx = 2\int \frac{1}{x+1}dx + 3\int \frac{1}{x-3}dx \\ &= 2\int \frac{(x+1)'}{x+1}dx + 3\int \frac{(x-3)'}{x-3}dx = 2\ln|x+1| + 3\ln|x-3| + c, \quad c \text{ ជាចំនួនពិតថេរ ។} \end{aligned}\]

ឧទាហរណ៍ 3 : សរសេរប្រភាគ \(\displaystyle \frac{6x+7}{(x+2)^{2}}\) ជាផលបូកនៃប្រភាគងាយ រួចគណនា \(\displaystyle \int \frac{6x+7}{(x+2)^{2}}dx\) គេមាន \(\displaystyle \frac{6x+7}{(x+2)^{2}} = \frac{A}{x+2} + \frac{B}{(x+2)^{2}} = \frac{Ax+2A+B}{(x+2)^{2}}\) នោះ \(\displaystyle 6x+7 = Ax+2A+B\) ដោយផ្ទឹមលេខមេគុណ គេបាន \(\displaystyle \begin{cases} A=6 \\ 2A+B=7 \end{cases}\) គេបាន \(\displaystyle A=6, B=-5\)

\[\begin{aligned} \int \frac{6x+7}{(x+2)^{2}}dx &= \int \left[\frac{6}{x+2} - \frac{5}{(x+2)^{2}}\right]dx = 6\int \frac{1}{x+2}dx - 5\int \frac{1}{(x+2)^{2}}dx \\ &= 6\int \frac{(x+2)'}{x+2}dx - 5\int \frac{(x+2)'}{(x+2)^{2}}dx = 6\ln|x+2| + \frac{5}{x+2} + c, \quad c \text{ ជាចំនួនពិតថេរ ។} \end{aligned}\]

គណនា \(\displaystyle \int \frac{dx}{x^{2}+x-2}\) ។

ដំណោះស្រាយ

គេមាន \(\displaystyle \frac{1}{x^{2}+x-2} = \frac{1}{(x-1)(x+2)} = \frac{A_{1}}{x-1} + \frac{A_{2}}{x+2} = \frac{A_{1}(x+2)+A_{2}(x-1)}{(x-1)(x+2)} = \frac{(A_{1}+A_{2})x + (2A_{1}-A_{2})}{(x-1)(x+2)}\) នោះគេបាន \(\displaystyle 1 = (A_{1}+A_{2})x + (2A_{1}-A_{2})\) គេបាន \(\displaystyle \begin{cases} A_{1}+A_{2}=0 \\ 2A_{1}-A_{2}=1 \end{cases} \implies 3A_{1}=1 \implies A_{1}=\frac{1}{3}, A_{2}=-\frac{1}{3}\) នោះ \(\displaystyle \frac{1}{x^{2}+x-2} = \frac{\frac{1}{3}}{x-1} + \frac{-\frac{1}{3}}{x+2}\) គេបាន

\[\begin{aligned} \int \frac{dx}{x^{2}+x-2} &= \int \left[\frac{\frac{1}{3}}{x-1} + \frac{-\frac{1}{3}}{x+2}\right]dx = \frac{1}{3}\int \frac{1}{x-1}dx - \frac{1}{3}\int \frac{1}{x+2}dx \\ &= \frac{1}{3}\int \frac{(x-1)'}{x-1}dx - \frac{1}{3}\int \frac{(x+2)'}{x+2}dx \\ &= \frac{1}{3}\ln|x-1| - \frac{1}{3}\ln|x+2| + c, \quad c \text{ ជាចំនួនពិតថេរ} \end{aligned}\]

ដូច្នេះ \(\displaystyle \int \frac{dx}{x^{2}+x-2} = \frac{1}{3}\ln|x-1| - \frac{1}{3}\ln|x+2| + c, c\) ជាចំនួនពិតថេរ ។

ដំណោះស្រាយ

គេមាន \(\displaystyle \frac{2x+4}{x^{3}-2x^{2}} = \frac{2x+4}{x^{2}(x-2)} = \frac{A}{x} + \frac{B}{x^{2}} + \frac{C}{x-2} = \frac{Ax(x-2)+B(x-2)+Cx^{2}}{x^{2}(x-2)} = \frac{(A+C)x^{2}+(-2A+B)x-2B}{x^{2}(x-2)}\) នោះគេបាន \(\displaystyle 2x+4 = (A+C)x^{2}+(-2A+B)x-2B\) ដោយផ្ទឹមលេខមេគុណ

\[\begin{cases} A+C=0 \\ -2A+B=2 \\ -2B=4 \end{cases} \implies \begin{cases} C=2 \\ A=-2 \\ B=-2 \end{cases}\]

គេបាន

\[\begin{aligned} \int \frac{2x+4}{x^{3}-2x^{2}}dx &= \int \left[\frac{-2}{x} + \frac{-2}{x^{2}} + \frac{2}{x-2}\right]dx = -2\int \frac{1}{x}dx - 2\int \frac{1}{x^{2}}dx + 2\int \frac{1}{x-2}dx \\ &= -2\int \frac{1}{x}dx - 2\int \frac{1}{x^{2}}dx + 2\int \frac{(x-2)'}{x-2}dx \\ &= -2\ln|x| + \frac{2}{x} + 2\ln|x-2| + c, \quad c \text{ ជាចំនួនពិតថេរ} \end{aligned}\]

ដូច្នេះ \(\displaystyle \int \frac{2x+4}{x^{3}-2x^{2}}dx = -2\ln|x| + \frac{2}{x} + 2\ln|x-2| + c, c\) ជាចំនួនពិតថេរ ។

ដំណោះស្រាយ

គេបាន \(\displaystyle \frac{x^{3}}{x^{2}+2x+1} = x - 2 + \frac{3x+2}{x^{2}+2x+1} = x - 2 + \frac{3x+2}{(x+1)^{2}} = x - 2 + \frac{A}{x+1} + \frac{B}{(x+1)^{2}}\) ដែល \(\displaystyle \frac{3x+2}{(x+1)^{2}} = \frac{A}{x+1} + \frac{B}{(x+1)^{2}} = \frac{Ax+A+B}{(x+1)^{2}}\) ។ គេបាន \(\displaystyle 3x+2 = Ax+A+B\) ដោយផ្ទឹមលេខមេគុណ គេបាន \(\displaystyle A=3, B=-1\) នោះ \(\displaystyle \frac{x^{3}}{x^{2}+2x+1} = x - 2 + \frac{3}{x+1} - \frac{1}{(x+1)^{2}}\) គេបាន

ឧទាហរណ៍ 1 : ស្ថានីយវិទ្យុមួយកំពុងតែចាប់ផ្តើមបង្កើនការផ្សព្វផ្សាយពាណិជ្ជកម្មដើម្បីបង្កើនចំនួនអ្នកស្តាប់ប្រចាំថ្ងៃ ។ បច្ចុប្បន្ននេះ អ្នកស្តាប់វិទ្យុនេះជាប្រចាំមានចំនួន \(\displaystyle 27\,000\) នាក់ ហើយគេសង្ឃឹមថាក្នុងមួយថ្ងៃចំនួនអ្នកស្តាប់ \(\displaystyle S(t)\) នឹងកើនឡើងតាមអត្រា \(\displaystyle S'(t) = 60\sqrt{t}\) ដែល \(\displaystyle t\) ជាចំនួនថ្ងៃគិតពីថ្ងៃចាប់ផ្តើមបង្កើនការផ្សព្វផ្សាយ ។ តើគេត្រូវប្រើរយៈពេលប៉ុន្មាន ប្រសិនបើស្ថានីយវិទ្យុនេះចង់ឱ្យចំនួនអ្នកស្តាប់ប្រចាំថ្ងៃកើនឡើងរហូតដល់ \(\displaystyle 41\,000\) នាក់?

សម្រាយបញ្ជាក់ : គេត្រូវដោះស្រាយសមីការ \(\displaystyle S(t) = 41\,000\) ដើម្បីទទួលបាន \(\displaystyle t\) ថ្ងៃ តាមបម្រាប់ គេមាន \(\displaystyle S'(t) = 60\sqrt{t}\) និង \(\displaystyle S(0) = 27\,000\) បើ \(\displaystyle S'(t) = 60\sqrt{t}\) នោះ \(\displaystyle S(t) = \int S'(t)dt = \int 60t^{\frac{1}{2}}dt = \frac{60t^{\frac{3}{2}}}{\frac{3}{2}} + c = 40t^{\frac{3}{2}} + c\) ដែល \(\displaystyle c\) ជាចំនួនពិតថេរ ។ ដោយ \(\displaystyle S(0) = 27\,000\) នោះគេបាន \(\displaystyle 40 \times 0^{\frac{3}{2}} + c = 27\,000 \implies c = 27\,000\) ។ នោះគេបាន \(\displaystyle S(t) = 40t^{\frac{3}{2}} + 27\,000\) ។ តាមសមីការ \(\displaystyle S(t) = 41\,000\) គេបាន \(\displaystyle 40t^{\frac{3}{2}} + 27\,000 = 41\,000 \implies 40t^{\frac{3}{2}} = 14\,000 \implies t^{\frac{3}{2}} = 350 \implies t = 49.66 \approx 50\) ។ ដូច្នេះ គេត្រូវប្រើពេល \(\displaystyle 50\) ថ្ងៃ ។
